CREATE  PROCEDURE [dbo].[SPGetAllDataByTableNameByDateTime]
@TableName nvarchar(50) = NULL,
@StartDateTime nvarchar(50) = NULL,
@EndDateTime nvarchar(50) = NULL
AS

SET @StartDateTime = ISNULL(@StartDateTime, GETDATE())
SET @EndDateTime = ISNULL(@EndDateTime, GETDATE())

BEGIN

DECLARE @strQry varchar(max) = '
SELECT 
(SELECT  * 
FROM ' + @TableName + '
WHERE CreatedOn > '''+ @StartDateTime + ''' AND CreatedOn <=''' + @EndDateTime + ''' 
FOR JSON PATH,
INCLUDE_NULL_VALUES) JsonData '


--PRINT @strQry
EXECUTE(@strQry)

END
